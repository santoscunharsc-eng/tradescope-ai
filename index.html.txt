    <script>
        function executeEngine() {
            const input = document.getElementById('rawInput').value;
            const btn = document.getElementById('actionBtn');
            const resultContainer = document.getElementById('resultContainer');
            const outputText = document.getElementById('outputText');

            if (!input.trim()) {
                alert('Please insert a raw request first.');
                return;
            }

            btn.disabled = true;
            btn.innerText = 'Analyzing linguistic vectors & building SOW...';

            setTimeout(() => {
                outputText.innerText = `[OFFICIAL STATEMENT OF WORK - SOW]
--------------------------------------------------
1. PROJECT INTENT & CONTEXT:
Processed Input: "${input}"
Detected Intent: Urgent commercial/residential custom fabrication request requiring structural precision.

2. TECHNICAL SPECIFICATIONS & DELIVERABLES:
- Core Architecture: Custom modular assembly engineered for high-durability integration.
- Surface & Finishing: Premium architectural finish matching international quality benchmarks.
- Tolerance & Compliance: Built to standard North American/European spatial tolerances.

3. COMMERCIAL ESTIMATE & TIMELINE:
- Production Window: 4 to 6 weeks from approval.
- Estimated Project Valuation: $2,200.00 - $4,500.00 USD.
--------------------------------------------------
Status: Ready for client submission.`;

                resultContainer.classList.remove('hidden');
                btn.disabled = false;
                btn.innerText = 'Generate Professional SOW & Scope';
                resultContainer.scrollIntoView({ behavior: 'smooth' });
            }, 1200);
        }

        function copyOutput() {
            const text = document.getElementById('outputText').innerText;
            navigator.clipboard.writeText(text);
            alert('Scope copied successfully!');
        }
